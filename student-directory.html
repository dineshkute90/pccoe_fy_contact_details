// Fetch Records Securely from Cloud Firestore async function fetchStudentsFromFirebase() { document.getElementById('dataStatus').innerText = 'Loading…'; try { // Fetch snapshot from Firestore const snapshot = await db.collection('contacts').get(); allStudents = []; snapshot.forEach((doc) => { let item = doc.data(); allStudents.push({ prn: item.PRN || '', name: item['Student Name'] || '', division: item.Division || '', batch: item.Batch || '', program: item.Program || '', photoUrl: item.Photo || '', studentMobile: item['Student Mobile'] || '', parentMobile: item['Parent Mobile'] || '', emailId: item['Email ID'] || '' }); }); // 1. Sort the records by PRN descending (e.g. 125B1... first, then 124B1...) allStudents.sort((a, b) => b.prn.localeCompare(a.prn, undefined, { numeric: true, sensitivity: 'base' })); // 2. Assign serial index numbers after sorting allStudents.forEach((student, i) => { student.index = i + 1; }); document.getElementById('totalStudents').innerText = allStudents.length; document.getElementById('dataStatus').innerText = 'Connected'; initDataTable(allStudents); } catch (error) { console.error("Error loading data:", error); document.getElementById('dataStatus').innerText = 'Error'; } } function initDataTable(data) { if (dataTable) { dataTable.destroy(); } const tbody = document.querySelector('#studentTable tbody'); tbody.innerHTML = ''; data.forEach(student => { const tr = document.createElement('tr'); tr.id = `row-${student.prn}`; if (selectedStudents.has(student.emailId)) { tr.classList.add('row-selected'); } const isChecked = selectedStudents.has(student.emailId) ? 'checked' : ''; const imgTag = student.photoUrl ? `` : `No Photo`; tr.innerHTML = ` ${student.index} ${student.prn} ${student.name} ${student.division} ${student.batch} ${student.program} ${imgTag} ${student.studentMobile} ${student.parentMobile} ${student.emailId} `; tbody.appendChild(tr); }); // Initialize DataTable with PRN column (index 2) sorted descending by default dataTable = $('#studentTable').DataTable({ pageLength: 10, responsive: true, order: [[2, 'desc']], // Column index 2 is PRN columnDefs: [ { orderable: false, targets: [0, 7] } // Disable sorting on checkbox & photo ] }); }
